Get started

Open source under MIT. Four public packages on npm — no token, no account. Install them and import one stylesheet. 

npm i @gener8v/ui @gener8v/tokens @gener8v/icons
import '@gener8v/ui/styles.css';
import { Button, Table, Dialog } from '@gener8v/ui';
